find_package(Qt6 REQUIRED COMPONENTS DBus Multimedia)
add_library(karyo-player-window STATIC src/library.cpp src/library.h
    src/playerwidgets.cpp src/playerwidgets.h src/playerwindow.cpp src/playerwindow.h)
target_link_libraries(karyo-player-window PUBLIC Karyo::Style karyo-app-filemanager Qt6::DBus Qt6::Multimedia)
target_include_directories(karyo-player-window PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
qt_add_executable(karyo-player src/main.cpp)
target_link_libraries(karyo-player PRIVATE karyo-player-window)
karyo_add_translation(karyo-player "${CMAKE_CURRENT_SOURCE_DIR}/translations/karyo-player_ru.ts")
install(TARGETS karyo-player RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT player-runtime)
install(FILES data/karyo-player.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications COMPONENT player-runtime)
install(FILES ${PROJECT_SOURCE_DIR}/LICENSE ${PROJECT_SOURCE_DIR}/docs/PROVENANCE.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/karyo-player COMPONENT player-runtime)
configure_file(data/player.in "${CMAKE_CURRENT_BINARY_DIR}/player" @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/player" DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT player-runtime)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    add_executable(karyo-player-test tests/playertest.cpp)
    target_link_libraries(karyo-player-test PRIVATE karyo-player-window Qt6::Test)
    target_compile_definitions(karyo-player-test PRIVATE KARYO_PLAYER_EXECUTABLE="$<TARGET_FILE:karyo-player>")
    add_dependencies(karyo-player-test karyo-player)
    add_test(NAME player-library-decode-ipc COMMAND dbus-run-session -- $<TARGET_FILE:karyo-player-test>)
    set_tests_properties(player-library-decode-ipc PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FFMPEG_ENCODING_HW_DEVICE_TYPES=;QT_FFMPEG_DECODING_HW_DEVICE_TYPES=")
endif()
