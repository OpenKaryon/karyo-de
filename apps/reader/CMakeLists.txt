find_package(Qt6 REQUIRED COMPONENTS DBus Pdf PdfWidgets)
add_library(karyo-reader-window STATIC src/documentpage.cpp src/documentpage.h
    src/readerwindow.cpp src/readerwindow.h src/thumbnails.cpp src/thumbnails.h)
target_link_libraries(karyo-reader-window PUBLIC Karyo::Style Qt6::DBus Qt6::Pdf Qt6::PdfWidgets)
target_include_directories(karyo-reader-window PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
qt_add_executable(karyo-reader src/main.cpp)
target_link_libraries(karyo-reader PRIVATE karyo-reader-window)
karyo_add_translation(karyo-reader "${CMAKE_CURRENT_SOURCE_DIR}/translations/karyo-reader_ru.ts")
install(TARGETS karyo-reader RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT reader-runtime)
install(FILES data/karyo-reader.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications COMPONENT reader-runtime)
install(FILES ${PROJECT_SOURCE_DIR}/LICENSE ${PROJECT_SOURCE_DIR}/docs/PROVENANCE.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/karyo-reader COMPONENT reader-runtime)
configure_file(data/pdf.in "${CMAKE_CURRENT_BINARY_DIR}/pdf" @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/pdf" DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT reader-runtime)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(karyo-reader-test tests/readertest.cpp)
    target_link_libraries(karyo-reader-test PRIVATE karyo-reader-window Qt6::Test)
    target_compile_definitions(karyo-reader-test PRIVATE KARYO_READER_EXECUTABLE="$<TARGET_FILE:karyo-reader>")
    add_dependencies(karyo-reader-test karyo-reader)
    add_test(NAME reader-document-ipc COMMAND dbus-run-session -- $<TARGET_FILE:karyo-reader-test>)
    set_tests_properties(reader-document-ipc PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
