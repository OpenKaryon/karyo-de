find_package(Qt6 REQUIRED COMPONENTS DBus)
add_library(karyo-files-views STATIC
    src/folderview.cpp src/folderview.h src/detailspane.cpp src/detailspane.h)
target_include_directories(karyo-files-views PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(karyo-files-views PUBLIC Karyo::Style karyo-file-presentation)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(karyo-files-rename-test tests/renametest.cpp)
    target_link_libraries(karyo-files-rename-test PRIVATE karyo-files-views Qt6::Test)
    add_test(NAME files-inline-rename COMMAND karyo-files-rename-test)
    set_tests_properties(files-inline-rename PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
    add_executable(karyo-files-details-test tests/detailstest.cpp)
    target_link_libraries(karyo-files-details-test PRIVATE karyo-files-views Qt6::Test)
    add_test(NAME files-folder-size-cache COMMAND karyo-files-details-test)
    set_tests_properties(files-folder-size-cache PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()

add_library(karyo-files-places STATIC src/places.cpp src/places.h)

target_link_libraries(karyo-files-places PUBLIC Karyo::Style Qt6::DBus)
if(BUILD_TESTING)
    add_executable(karyo-files-places-test tests/placestest.cpp)
    target_link_libraries(karyo-files-places-test PRIVATE karyo-files-places Qt6::Test)
    add_test(NAME files-places-usb COMMAND dbus-run-session -- /bin/sh
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/run-private-bus.sh $<TARGET_FILE:karyo-files-places-test>)
    set_tests_properties(files-places-usb PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()
