find_package(Qt6 6.4 REQUIRED COMPONENTS Multimedia MultimediaWidgets)
qt_add_executable(karyo-camera src/main.cpp src/camerawindow.cpp src/camerawindow.h)
target_link_libraries(karyo-camera PRIVATE Karyo::Style Qt6::Multimedia Qt6::MultimediaWidgets)
karyo_add_translation(karyo-camera "${CMAKE_CURRENT_SOURCE_DIR}/translations/karyo-camera_ru.ts")
install(TARGETS karyo-camera RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT camera-runtime)
karyo_install_desktop_entry(data/karyo-camera.desktop camera-runtime)
install(FILES ${PROJECT_SOURCE_DIR}/LICENSE ${PROJECT_SOURCE_DIR}/docs/PROVENANCE.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/karyo-camera COMPONENT camera-runtime)
configure_file(data/camera.in "${CMAKE_CURRENT_BINARY_DIR}/camera" @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/camera" DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT camera-runtime)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(karyo-camera-test tests/cameratest.cpp src/camerawindow.cpp src/camerawindow.h)
    target_include_directories(karyo-camera-test PRIVATE src)
    target_link_libraries(karyo-camera-test PRIVATE Karyo::Style Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Test)
    add_test(NAME camera-no-devices COMMAND karyo-camera-test)
    set_tests_properties(camera-no-devices PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
