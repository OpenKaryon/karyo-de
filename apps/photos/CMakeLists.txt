find_package(Qt6 REQUIRED COMPONENTS DBus)
qt_add_executable(karyo-photos src/main.cpp src/photoswindow.cpp src/photoswindow.h
    src/gallery.cpp src/gallery.h src/imageview.cpp src/imageview.h src/thumbnails.cpp src/thumbnails.h)
target_link_libraries(karyo-photos PRIVATE Karyo::Style karyo-app-filemanager Qt6::DBus)
karyo_add_translation(karyo-photos "${CMAKE_CURRENT_SOURCE_DIR}/translations/karyo-photos_ru.ts")
install(TARGETS karyo-photos RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT photos-runtime)
karyo_install_desktop_entry(data/karyo-photos.desktop photos-runtime)
install(FILES ${PROJECT_SOURCE_DIR}/LICENSE ${PROJECT_SOURCE_DIR}/docs/PROVENANCE.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/karyo-photos COMPONENT photos-runtime)
configure_file(data/viewer.in "${CMAKE_CURRENT_BINARY_DIR}/viewer" @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/viewer" DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT photos-runtime)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(karyo-photos-test tests/imagetest.cpp src/imageview.cpp src/imageview.h)
    target_include_directories(karyo-photos-test PRIVATE src)
    target_link_libraries(karyo-photos-test PRIVATE Qt6::Widgets Qt6::Test)
    add_test(NAME photos-image-rotation COMMAND karyo-photos-test)
    set_tests_properties(photos-image-rotation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
