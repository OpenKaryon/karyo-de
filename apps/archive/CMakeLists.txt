find_package(LibArchive REQUIRED)
# Files reuses this backend from a sibling directory; Qt finalizers walk its imports.
set_property(TARGET LibArchive::LibArchive PROPERTY IMPORTED_GLOBAL TRUE)
add_library(karyo-archive-backend STATIC src/archivebackend.cpp src/archivebackend.h
    src/archive7zip.cpp src/archive7zip.h)
target_include_directories(karyo-archive-backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(karyo-archive-backend PUBLIC Qt6::Core LibArchive::LibArchive)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(karyo-archive-backend-test tests/backendtest.cpp)
    target_link_libraries(karyo-archive-backend-test PRIVATE karyo-archive-backend Qt6::Test)
    target_compile_definitions(karyo-archive-backend-test PRIVATE ARCHIVE_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME archive-backend COMMAND karyo-archive-backend-test)
    set_tests_properties(archive-backend PROPERTIES TIMEOUT 60)
endif()

add_library(karyo-archive-window STATIC src/archivewindow.cpp src/archivewindow.h
    src/archivepassworddialog.cpp src/archivepassworddialog.h)
target_link_libraries(karyo-archive-window PUBLIC karyo-archive-backend Karyo::Style karyo-file-presentation karyo-app-filemanager)
qt_add_executable(karyo-archive src/main.cpp)
target_link_libraries(karyo-archive PRIVATE karyo-archive-window)
karyo_add_translation(karyo-archive "${CMAKE_CURRENT_SOURCE_DIR}/translations/karyo-archive_ru.ts")
install(TARGETS karyo-archive RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT archive-runtime)
karyo_install_desktop_entry(data/karyo-archive.desktop archive-runtime)
install(FILES data/karyo-archive.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps COMPONENT archive-runtime)
install(FILES ${PROJECT_SOURCE_DIR}/LICENSE ${PROJECT_SOURCE_DIR}/docs/PROVENANCE.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/karyo-archive COMPONENT archive-runtime)
configure_file(data/archive.in "${CMAKE_CURRENT_BINARY_DIR}/archive" @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/archive" DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT archive-runtime)
install(FILES data/rar_addon.md ${PROJECT_SOURCE_DIR}/LICENSE
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/karyo-archive-rar COMPONENT archive-rar-addon)
if(BUILD_TESTING)
    add_executable(karyo-archive-window-test tests/windowtest.cpp)
    target_link_libraries(karyo-archive-window-test PRIVATE karyo-archive-window Qt6::Test)
    add_test(NAME archive-window-passwords COMMAND karyo-archive-window-test)
    set_tests_properties(archive-window-passwords PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()
